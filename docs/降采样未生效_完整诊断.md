如果没有 `max_resolution` 参数，需要手动添加。完整的函数应该是：

```python
def _decode_frames(mp4_bytes: bytes, max_resolution: int = 640) -> np.ndarray | None:
    """解码视频帧，自动降采样到指定分辨率

    Args:
        mp4_bytes: 视频文件字节
        max_resolution: 最大边长（默认 640，约 480p）
    """
    if not mp4_bytes:
        return None
    try:
        import av
        frames = []
        with av.open(io.BytesIO(mp4_bytes)) as c:
            for pkt in c.demux(video=0):
                for f in pkt.decode():
                    frame = f.to_ndarray(format="rgb24")

                    # 降采样到 max_resolution（如果指定）
                    if max_resolution is not None:
                        H, W = frame.shape[:2]
                        if max(H, W) > max_resolution:
                            import cv2
                            scale = max_resolution / max(H, W)
                            new_H, new_W = int(H * scale), int(W * scale)
                            frame = cv2.resize(frame, (new_W, new_H), interpolation=cv2.INTER_AREA)

                    frames.append(frame)
        return np.array(frames, dtype=np.uint8) if frames else None
    except Exception:
        return None
```
